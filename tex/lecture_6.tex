\section{A new prediction workflow}
\begin{frame}{A pretrained model can learn a new table in context}
\small
\begin{columns}[c]
\column{.48\textwidth}
Traditional supervised learning fits task-specific parameters from labeled rows. A tabular foundation model (TFM) first learns across many tables, then uses a new table's labeled rows as context for predictions.
\begin{block}{\textbf{At inference:}}
Supply labeled context and unlabeled query rows. The pretrained weights stay frozen; the model conditions its prediction on the context.
\end{block}
Fine-tuning, task-specific, and combination models are also possible, but are separate procedures.
\column{.48\textwidth}\centering

\node[b,fill=blue!8](pre){many training tasks $\longrightarrow$ pretrained weights $\theta$};
\node[b,fill=green!10,below=of pre](ctx){new task: labeled context $D_C=(X_C,y_C)$};
\node[b,fill=orange!12,below=of ctx](q){query features $X_Q$};

\end{tikzpicture}
\visualdescription{Pretraining sets weights; labeled examples in the new task condition a frozen model at prediction time.}
\end{columns}
\slidesource{Müller et al. (2022), \href{https://openreview.net/forum?id=KSugKcbNf9}{\emph{Transformers Can Do Bayesian Inference};} Qu et al. (2025), \href{https://proceedings.mlr.press/v267/qu25d.html}{\emph{TabICL}}}
\end{frame}

\begin{frame}{Tabular foundation models use different transfer strategies}
\small
\begin{tabularx}{\linewidth}{@{}p{.21\linewidth}p{.31\linewidth}Y@{}}

\midrule

\begin{block}{\textbf{Distinct tasks:}}
\textbf{Pretraining} updates shared model weights across tasks. \textbf{In-context learning} uses labeled examples at inference without weight updates. \textbf{Fine-tuning} updates weights for a downstream task.
\end{block}
\slidesource{Hollmann et al. (2023), \href{https://arxiv.org/abs/2207.01848}{TabPFN}; Qu et al. (2025), \href{https://proceedings.mlr.press/v267/qu25d.html}{TabICL}; Ma et al. (2025), \href{https://proceedings.neurips.cc/paper_files/paper/2025/file/fc0e3f908a2116ba529ad0a1530a3675-Paper-Conference.pdf}{TabDPT}}
\end{frame}

\section{Kumo Tabular architecture}
\begin{frame}{Kumo turns each cell into a feature-group token}
\small
\begin{columns}[c]
\column{.53\textwidth}
For a cell value $x_{ij}$ in row $i$, feature $j$, a learned Fourier map provides a cell encoding of width $d_{\text{cell}}$:
\[
{\scriptstyle \phi(x_{ij})=[\sin(\omega_1x_{ij}),\cos(\omega_1x_{ij}),\ldots,\sin(\omega_Fx_{ij}),\cos(\omega_Fx_{ij})]}
\]
Overlapping groups of three cell encodings are concatenated, then a learned projection maps each group to a Phase 1 token of width $d_{\text{model}}$. A label embedding is added for labeled context rows for phases 1 and 2 of Kumo.
\column{.44\textwidth}\centering
\begin{tikzpicture}[every node/.style={font=\scriptsize,align=center},c/.style={draw,minimum width=12mm,minimum height=7mm}]
\node[c,fill=blue!10](x1){$x_1$};\node[c,right=0pt of x1,fill=blue!10](x2){$x_2$};\node[c,right=0pt of x2,fill=blue!10](x3){$x_3$};\node[c,right=0pt of x3,fill=blue!10](x4){$x_4$};\node[c,right=0pt of x4,fill=blue!10](x5){$x_5$};
\draw[very thick,brand,rounded corners]($(x1.north west)+(-1mm,1mm)$)rectangle($(x3.south east)+(1mm,-1mm)$);
\draw[very thick,orange,rounded corners]($(x3.north west)+(-1mm,1mm)$)rectangle($(x5.south east)+(1mm,-1mm)$);
\node[below=1mm of x3](g){overlapping groups};
\node[below=of g,draw,fill=green!10,text width=52mm](e){cell Fourier encodings $\rightarrow$\\group concatenation $\rightarrow$\\project to $d_{\text{model}}$};
\draw[->](g)--(e);
\end{tikzpicture}
\visualdescription{Overlapping groups concatenate cell encodings before a learned projection produces a $d_{\text{model}}$-wide Phase 1 token.}
\end{columns}
\medskip
Example dimensions: three 16-wide cell encodings concatenate into a 48-wide feature-group input, then a learned projection produces a 32-wide Phase 1 token ($d_{\text{model}}=32$).
\slidesource{NVIDIA, \href{https://huggingface.co/blog/nvidia/kumo-tabular}{\emph{Kumo Tabular architecture overview}}; model details, \href{https://docs.nvidia.com/sdgm/structured-data-models/models/kumo-tabular}{official documentation}}
\end{frame}

\begin{frame}{Attention routes information with learned queries, keys, and values}
\small
\begin{columns}[c]
\column{.46\textwidth}

Given vectors $x_i$, learned projections produce $q_i=W_Qx_i$, $k_i=W_Kx_i$, $v_i=W_Vx_i$; $d_k$ is the key-vector dimension.\\

\medskip
Attention weights compare a query to keys:
\[
\alpha_{ij}=\operatorname{softmax}_j(q_i^\top k_j/\sqrt{d_k}),\quad o_i=\sum_j\alpha_{ij}v_j.
\]
\medskip
{\scriptsize\color{sourcegray}\textbf{Note:} Query means attention vector, not query row.}
\column{.51\textwidth}\centering
\begin{tikzpicture}[>=Latex,node distance=7mm,every node/.style={font=\scriptsize,align=center},b/.style={draw,rounded corners,text width=19mm,minimum height=8mm}]
\node[b,fill=gray!10](in){input\\vectors $x_j$};
\node[b,fill=blue!10,below left=of in](q){query $q_i$};\node[b,fill=green!10,below=of in](k){keys $k_j$};\node[b,fill=orange!12,below right=of in](v){values $v_j$};
\node[b,fill=purple!10,below=13mm of k](out){weighted sum $o_i$};
\draw[->](in)--(q);\draw[->](in)--(k);\draw[->](in)--(v);\draw[->](q)--(out);\draw[->](k)--(out);\draw[->](v)--(out);
\end{tikzpicture}
\visualdescription{Queries choose which values to combine by matching against keys. Outputs carry the selected values to the next layer.}
\end{columns}
\slidesource{Vaswani et al. (2017), \href{https://arxiv.org/abs/1706.03762}{\emph{Attention Is All You Need}}}
\end{frame}

\begin{frame}{How attention selects and carries information}
\footnotesize
\begin{columns}[c]
\column{.54\textwidth}
\begin{itemize}\itemsep.25em\scriptsize
\item For each representation $x_i$, the model computes a query $q_i = W_Q x_i$.
\item For each representation $x_j$, it computes a key $k_j = W_K x_j$.
\item The dot product $q_i^\top k_j$ is a learned compatibility score; larger scores receive larger attention weights.
\item The value $v_j = W_V x_j$ is the information carried by token $j$, with learned adjustments. 
\item During training, $W_Q$, $W_K$, and $W_V$ are adjusted so attention can favor information useful for the model's objective.
\item The attention output $o_i$ is transformed by an output projection, then added to $x_i$ through a residual connection before further processing.
\end{itemize}
\column{.43\textwidth}
\begin{block}{Intuition}\scriptsize
Think of $q_i$ as a signal about what token $i$ needs, and $k_j$ as a signal about what another token $j$ can offer. The value $v_j$ is the learned information from token $j$ that attention can pass along. A stronger match between $q_i$ and $k_j$ gives $v_j$ more weight in the ongoing transformation process. The output $o_i$ combines these weighted values.\\
\vspace{8pt}
A later step combines this result with the original representation $x_i$ through a residual connection, helping preserve the input signal as it passes through deeper layers. Subsequent layers continue transforming token representations into forms useful for predicting the desired label or outcome.
\end{block}
\end{columns}
\slidesource{Vaswani et al. (2017), \href{https://arxiv.org/abs/1706.03762}{\emph{Attention Is All You Need}}}
\end{frame}

\begin{frame}{Kumo alternates attention across features and across rows}
\small
\begin{columns}[c]
\column{.48\textwidth}
\begin{block}{\textbf{Row attention:}}
Within each row, feature-group tokens attend across groups. This lets representations combine signals from different columns and learn relationships among them.
\end{block}
\begin{block}{\textbf{Column attention:}}
At a feature-group position, rows exchange information through learned summary (``inducing'') vectors, helping the model learn the column's value distribution. Only context rows are information sources; query rows do not contribute keys or values.
\end{block}

\begin{frame}{Pretraining learns the context-to-query prediction procedure}
\small
{\footnotesize
\begin{enumerate}\itemsep.55em
\item Sample a training table and divide rows into context and query sets. Supply context labels; withhold query labels.
\item Process both sets through the row encoder and in-context prediction stages; produce query predictions.
\item Compare predictions with withheld targets, compute loss, and backpropagate. A gradient-based optimizer updates all trainable components: cell/group encoders (including Fourier frequencies and projections), label embeddings, CLS and summary (``inducing'') vectors, attention projections and learned scaling/gates, normalization parameters, FFNs, any projection between phases, and the prediction head.
\item Repeat over many sampled tables, tasks, and splits.
\end{enumerate}
}
\begin{block}{\textbf{Frozen inference:}}
For a new task, provide labeled context and query features, then predict. There is no query-label loss or optimizer step; pretrained weights do not change.
\end{block}
\slidesource{Müller et al. (2022), \href{https://openreview.net/forum?id=KSugKcbNf9}{\emph{Transformers Can Do Bayesian Inference}}; Hollmann et al. (2023), \href{https://arxiv.org/abs/2207.01848}{TabPFN}}
\end{frame}

\section{Using and explaining predictions}
\begin{frame}{GAMs, GA$^2$Ms, EBMs, and NAMs share additive structure}
\small
\[
\operatorname{link}\{\mathbb{E}(Y\mid x)\}=\beta_0+\sum_j\beta_j g_j(x_j)+\sum_{j<k}\beta_{j,k}g_{j,k}(x_j,x_k)
\]
\begin{itemize}\itemsep.5em
\item A GAM with only $g_j$ has main effects; a GA$^2$M adds selected pairwise interactions $g_{j,k}$.
\item EBMs fit boosted shape functions (and optionally interactions); NAMs use a separate neural network per feature.
\item The notebook's Kumo-derived classifier is a main-effects GAM; the comparison NAM uses feature-specific networks.
\end{itemize}
Components describe model associations under the chosen inputs, not causes. Correlated predictors can share or shift apparent effects.
\slidesource{Caruana et al. (2015), \href{https://doi.org/10.1145/2783258.2788613}{\emph{Intelligible Models for HealthCare}}; Agarwal et al. (2021), \href{https://proceedings.neurips.cc/paper/2021/hash/251bd0442dfcc53b5a761e050f8022b8-Abstract.html}{NAM}; Lengerich et al. (2020), \href{https://proceedings.mlr.press/v108/lengerich20a.html}{functional ANOVA}}
\end{frame}

\begin{frame}{Frozen component-wise TFM predictions can also form an additive model}
\small
For each feature $j$, frozen Kumo receives only that feature and the same labeled context rows; it returns $p_j(x_j)$. Convert clipped probabilities to shape-function scores $g_j(x_j)=\operatorname{logit}(p_j(x_j))$, then fit a penalized logistic combiner on separate labeled validation data:
\[
\operatorname{logit}\hat p(y=1\mid x)=\beta_0+\sum_{j=1}^{P}\beta_j g_j(x_j).
\]
This is a Kumo-derived additive GAM.

\smallskip
For simplicity, the lecture-example notebook fits a GAM, but the same procedure could include pairwise feature functions to fit a potentially more accurate GA$^2$M. Its combiner standardizes fixed feature scores before fitting.
\slidesource{Hall (2026), \href{https://github.com/jphall663/GWU_LLM/blob/main/misc/existing_materials/lecture_6.ipynb}{Lecture 6: Neural Additive Models and Tabular Foundation Models notebook}; Lengerich et al. (2020), \href{https://proceedings.mlr.press/v108/lengerich20a.html}{functional ANOVA}; Agarwal et al. (2021), \href{https://proceedings.neurips.cc/paper/2021/hash/251bd0442dfcc53b5a761e050f8022b8-Abstract.html}{NAM}}
\end{frame}

\begin{frame}{Use validation data to choose settings; reserve test data for evaluation}
\small
\begin{columns}[c]
\column{.52\textwidth}
\textbf{Predictive settings:}
\begin{itemize}\itemsep.25em
\item Which labeled examples form context; context size.
\item Preprocessing and feature inclusion.
\item Number of randomized estimator views or ensemble members.
\end{itemize}
\textbf{Computational settings:}
\begin{itemize}\itemsep.25em
\item Query batch size trades memory for throughput.
\item Device, model size, and serving route affect feasibility and latency.
\end{itemize}
\column{.44\textwidth}
Kumo offers public pretrained small/medium/large checkpoints through \texttt{sdm.models.KumoTabular} and a hosted NVIDIA service. Sizes span about 27M--216M parameters. Memory and latency depend on context and query batch, so profile on the target device.
\begin{block}{Split discipline}
Validation selects settings or fits a combiner/calibrator. Test data estimates final performance once choices are locked.
\end{block}
\end{columns}

\begin{frame}{Repeated residual fitting can refine component shapes}
\scriptsize
\begin{columns}[c]
\column{.48\textwidth}
\begin{itemize}\itemsep.35em
\item Initialize predictions: $\hat y$.
\item Cycle through single features in column-index order: compute current residuals, fit $x_j$, and add its shape update to $\hat y$.
\item Screen interactions once with a tree/GBM; select the top 5 pairs and reuse them on later sweeps.
\item Cycle through those pairs: recompute residuals, fit each pair, and add its shape update to $\hat y$.
\item Repeat the entire sequence until convergence.
\item Repeated updates accumulate into main-effect and pairwise shape functions.
\item Cyclic residual fitting can capture patterns a one-shot score combiner misses.
\end{itemize}

\begin{frame}{Choose a TFM as a model to validate, not a default winner}
\footnotesize
\begin{columns}[c]
\column{.50\textwidth}
\begin{itemize}\itemsep.3em
\item Define the prediction task, data access, latency and hardware constraints.
\item Compare a TFM with strong task-appropriate baselines on leakage-safe splits; tabular data benchmarks are not authoritative for your task.
\item Test sensitivity to context selection, preprocessing, missingness, and context size.
\item Check calibration, subgroup behavior, shift, explanation stability, and operational cost.
\item Keep the final test set out of model and setting selection.
\end{itemize}

\begin{block}{\textbf{Takeaways:}}
TFMs move task adaptation into the prediction context. They do not remove the need for careful validation, data governance, or domain judgment.
\end{block}

\begin{frame}{Potential far-reaching governance implications of TFMs}
\small
\begin{itemize}\itemsep.55em
\item Reduce costly, inconsistent model building: individual discretion over complexity, tuning, and validation can produce avoidable variation in quality.
\item Make approved pretrained TFM families the default development pathway; require documented justification for departures.
\item Standardize documentation, validation, deployment, and monitoring around each approved TFM family.
\item For higher-risk applications, reuse pretrained predictors through inspectable GAM/GA$^2$M models with main effects and selected interactions.
\item Focus specialist effort on data quality, application suitability, independent validation, and ongoing monitoring rather than repeatedly building bespoke models.
\end{itemize}
\end{frame}
